cmake_minimum_required(VERSION 3.20)
project(SpearCollisionLab LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(Jolt CONFIG REQUIRED PATHS "${ROOT}/Intermediate/JoltMigration/Install/Development/lib/cmake/Jolt" NO_DEFAULT_PATH)
add_executable(SpearCollisionLab main.cpp)
target_link_libraries(SpearCollisionLab PRIVATE Jolt::Jolt)
target_compile_features(SpearCollisionLab PRIVATE cxx_std_20)
target_compile_options(SpearCollisionLab PRIVATE /fp:precise /GR- /EHsc)
add_custom_command(TARGET SpearCollisionLab POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:Jolt::Jolt> $<TARGET_FILE_DIR:SpearCollisionLab>)
