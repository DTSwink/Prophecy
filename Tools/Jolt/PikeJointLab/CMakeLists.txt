cmake_minimum_required(VERSION 3.20)
project(PikeJointLab LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(Jolt CONFIG REQUIRED PATHS "${ROOT}/Intermediate/JoltMigration/Install/Development/lib/cmake/Jolt" NO_DEFAULT_PATH)
add_executable(PikeJointLab main.cpp)
target_link_libraries(PikeJointLab PRIVATE Jolt::Jolt)
target_compile_features(PikeJointLab PRIVATE cxx_std_20)
target_compile_options(PikeJointLab PRIVATE /fp:precise /GR- /EHsc)
add_custom_command(TARGET PikeJointLab POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:Jolt::Jolt> $<TARGET_FILE_DIR:PikeJointLab>)
add_executable(SlashJointLab slash.cpp)
target_link_libraries(SlashJointLab PRIVATE Jolt::Jolt)
target_compile_features(SlashJointLab PRIVATE cxx_std_20)
target_compile_options(SlashJointLab PRIVATE /fp:precise /GR- /EHsc)
add_custom_command(TARGET SlashJointLab POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:Jolt::Jolt> $<TARGET_FILE_DIR:SlashJointLab>)
